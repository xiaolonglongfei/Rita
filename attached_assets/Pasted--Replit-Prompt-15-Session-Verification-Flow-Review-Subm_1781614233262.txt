## Overview

This prompt builds the complete Session Verification system and updates the Review submission flow to require session details before rating.

---

## Business Logic Summary

### Review Submission (Updated)
- Student fills in session details FIRST (required): date, time, location
- Then rates with vertical sliders (Value, Effectiveness, Punctuality)
- Then optional comment
- On submit: review saved as "Unverified", verification request sent to instructor
- One review per session per student (same student cannot submit two reviews for the same instructor on the same date)

### Instructor Verification
- Instructor logs in and sees pending verification requests
- Each request shows: date, time, location, student name (NOT the review scores or comment)
- Instructor clicks Confirm → review becomes "Verified"
- Instructor clicks Deny → review stays "Unverified" (not deleted)
- If instructor has not claimed profile → review stays "Unverified" permanently

### Notifications
- When student submits review → email sent to instructor (if instructor has email on file)
- Instructor also sees pending requests in their dashboard when logged in

### Display Rules (unchanged)
- Reviews only shown when instructor has 3+ reviews
- Student name NEVER shown on review cards
- Session date/time/location NEVER shown on review cards

## Step 4: Update Reviews API Route

Update `app/api/reviews/route.ts` to handle session details and duplicate prevention:

```typescript
import { createClient } from '@/lib/supabase/server'
import { NextResponse } from 'next/server'

export async function POST(request: Request) {
  const supabase = await createClient()

  const { data: { user } } = await supabase.auth.getUser()
  if (!user) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  }

  const body = await request.json()
  const {
    instructor_id,
    rating_value,
    rating_effectiveness,
    rating_punctuality,
    comment,
    session_date,
    session_time,
    session_location,
  } = body

  // Validate required fields
  if (!instructor_id || rating_value == null || 
      rating_effectiveness == null || rating_punctuality == null ||
      !session_date || !session_time || !session_location) {
    return NextResponse.json({ error: 'All session details are required' }, { status: 400 })
  }

  // Validate scores
  const scores = [rating_value, rating_effectiveness, rating_punctuality]
  if (scores.some((s: number) => s < 0 || s > 5)) {
    return NextResponse.json({ error: 'Scores must be between 0 and 5' }, { status: 400 })
  }

  // Prevent future session dates
  if (new Date(session_date) > new Date()) {
    return NextResponse.json({ error: 'Session date cannot be in the future' }, { status: 400 })
  }

  // Ensure user exists in users table
  await supabase.from('users').upsert({
    id: user.id,
    email: user.email!,
    full_name: user.user_metadata?.full_name || user.email!,
  }, { onConflict: 'id' })

  // Check for duplicate (same student, same instructor, same date)
  const { data: existing } = await supabase
    .from('reviews')
    .select('id')
    .eq('student_id', user.id)
    .eq('instructor_id', instructor_id)
    .eq('session_date', session_date)
    .single()

  if (existing) {
    return NextResponse.json({
      error: 'You have already submitted a review for a session with this instructor on this date'
    }, { status: 409 })
  }

  // Insert review
  const { data, error } = await supabase
    .from('reviews')
    .insert({
      student_id: user.id,
      instructor_id,
      rating_value,
      rating_effectiveness,
      rating_punctuality,
      comment: comment || null,
      session_date,
      session_time,
      session_location,
      is_verified: false,
      moderation_status: 'approved',
      edit_count: 0,
    })
    .select()
    .single()

  if (error) {
    return NextResponse.json({ error: error.message }, { status: 500 })
  }

  return NextResponse.json({ review: data }, { status: 201 })
}
```

## Step 6: Build Verification API Route

Create `app/api/reviews/[id]/verify/route.ts`:

```typescript
import { createClient } from '@/lib/supabase/server'
import { NextResponse } from 'next/server'

export async function POST(
  request: Request,
  { params }: { params: { id: string } }
) {
  const supabase = await createClient()

  const { data: { user } } = await supabase.auth.getUser()
  if (!user) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  }

  const body = await request.json().catch(() => ({}))
  const formData = await request.formData().catch(() => null)
  const action = body.action || formData?.get('action')

  if (action !== 'confirm' && action !== 'deny') {
    return NextResponse.json({ error: 'Invalid action' }, { status: 400 })
  }

  // Verify this instructor owns this review
  const { data: instructorProfile } = await supabase
    .from('instructors')
    .select('id')
    .eq('claimed_by', user.id)
    .single()

  if (!instructorProfile) {
    return NextResponse.json({ error: 'Not an instructor' }, { status: 403 })
  }

  const { data: review } = await supabase
    .from('reviews')
    .select('id, instructor_id')
    .eq('id', params.id)
    .eq('instructor_id', instructorProfile.id)
    .single()

  if (!review) {
    return NextResponse.json({ error: 'Review not found' }, { status: 404 })
  }

  if (action === 'confirm') {
    await supabase
      .from('reviews')
      .update({ is_verified: true })
      .eq('id', params.id)
  }
  // If deny: review stays, is_verified remains false

  return NextResponse.redirect(new URL('/sessions', request.url))
}
```
